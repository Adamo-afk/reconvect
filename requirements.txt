# IMPORTANT: On Windows this project needs Python 3.10 and a conda env.
# TensorFlow < 2.11 is one of the last versions with native Windows GPU
# support. Full install recipe (see README.md 'Installation' for prose):
#
#   1. Install CUDA 11.2 and cuDNN 8.1 at SYSTEM level via the NVIDIA
#      installers (NOT via conda). `conda install cudatoolkit=11.2
#      cudnn=8.1.0` silently downgrades Python 3.10 to 3.9 because
#      conda-forge has no CUDA 11.2 builds for Python 3.10.
#      After the CUDA install, also make sure the CUDA_PATH env var is
#      set (Python 3.8+ on Windows finds CUDA via CUDA_PATH, not PATH):
#          setx CUDA_PATH "C:\Program Files\NVIDIA GPU Computing Toolkit\CUDA\v11.2"
#      then open a fresh terminal.
#
#   2. Create the Python 3.10 conda env:
#          conda create -n tfenv python=3.10 -y
#          conda activate tfenv
#          python --version    # must print Python 3.10.x
#
#   3. Install the geospatial / cartography stack via conda-forge FIRST,
#      so pyproj, proj, geopandas, pyogrio, shapely, cartopy, pyresample
#      and pykdtree all share the same PROJ ABI. pip-installing pyproj
#      / geopandas on Windows produces mismatched proj.db locations and
#      every CRS lookup fails with "Invalid projection ... no database
#      context specified".
#
#      If you are RESCUING an env that previously pip-installed any of
#      these packages and you already ran `pip uninstall`, physically
#      delete the leftover site-packages\<pkg>\ directory before the
#      conda install. Otherwise conda sees the empty dir, skips writing
#      into it, and the package imports as an empty namespace package
#      (symptom: AttributeError: module 'pyproj' has no attribute 'CRS').
#          conda install -c conda-forge \
#              pyproj proj geopandas pyogrio shapely cartopy \
#              pyresample pykdtree \
#              netCDF4 xarray hdf5plugin h5py -y
#
#   4. Install the pip requirements below (TensorFlow, numpy, and
#      everything else that isn't in the geo stack):
#          pip install -r requirements.txt
#
#   5. Install the local c4dl package:
#          pip install -e .
#
# =============================================================================

# Deep learning
tensorflow<2.11

# Scientific computing
# numpy is pinned to 1.x because TensorFlow < 2.11 was compiled against
# the numpy 1.x C API. numpy 2.x raises "_ARRAY_API not found" on every
# TF import; leaving numpy unpinned means pip picks the latest and TF
# refuses to load. numpy >=1.23 is required by TF 2.10.
numpy<2,>=1.23
scipy
scikit-learn

# Geospatial / meteorological
# NOTE: on Windows the following must have been installed via
# `conda install -c conda-forge` BEFORE `pip install -r requirements.txt`
# runs (see header):
#     pyproj proj geopandas pyogrio shapely cartopy
#     pyresample pykdtree
#     netCDF4 xarray hdf5plugin h5py
# Pip will see they're already present in the env and skip them.
# Listing them here so pip can still resolve them on non-Windows
# platforms where a conda-forge preinstall isn't required.
pyresample
pyproj
netCDF4
xarray
geopandas        # read_kml_version2.py reads LINET KMLs via GeoPandas

# MTG FCI L1C data pipeline (pipeline_msg_mtg.py, inspect_mtg.py)
paramiko         # SFTP transfer from anm@192.168.11.223 — required by pipeline_msg_mtg.py
hdf5plugin       # CharLS plugin required to read FCI L1C compressed chunks

# OPERA radar pipeline (our_data/opera_data/)
h5py             # read OPERA ODC HDF5 files (reproject_opera.py)

# GPU acceleration for compute_normalization_stats.py.
# Match the wheel to the CUDA major already on disk; CUDA 11.2 (the
# version TensorFlow is pinned to here) needs the 11.x wheel.
cupy-cuda11x

# Visualization
matplotlib
plotly           # feature_importance_analysis.py — interactive Xi heatmap / bar chart

# Optional: high-quality country borders on the full-domain prediction
# plots and on identify_patches' diagnostic figures. Without cartopy
# those scripts fall back to a coarse hardcoded Romania polygon (no
# neighbour borders). Conda is strongly preferred on Windows because
# the pip wheels for cartopy's GEOS/PROJ dependencies are unreliable:
#     conda install -c conda-forge cartopy
cartopy

pandas
